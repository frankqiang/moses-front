**Before（硬编码）** ❌:
```vue
<script>
import { PLAN_STATUS_MAP, PLAN_STATUS_OPTIONS } from '../constants'

  data() {
    return {
      statusOptions: PLAN_STATUS_OPTIONS

    }
  }
```

```vue
<script>
import dictionaryMixin from '@/mixins/dictionaryMixin'

  mixins: [dictionaryMixin],
  // 自动可用：
  // - this.planStatusOptions
  // - this.getPlanStatusLabel(status)

```

### 动态配置模式（PlanSearch.vue）

```javascript
computed: {
  searchFormConfig() {
    return [
      {
        prop: 'status',
        type: 'select',
        options: this.planStatusOptions // 动态从字典获取
      }
    ]
  }
}
```

### 动态生成映射模式（PlanTable.vue）

```javascript
computed: {
  statusConfig() {
    // 从字典选项生成 textMap
    const textMap = {}
    if (this.planStatusOptions) {
      this.planStatusOptions.forEach(option => {
        textMap[option.value] = option.label
      })
    }
    return {
      textMap,
      typeMap: PLAN_STATUS_TYPE_MAP // 保留UI样式
    }
  }
}
```

## ✨ 技术亮点

### 1. 智能缓存机制

```javascript
// 三级加载策略
if (state.loaded && !forceRefresh) {
  return // 1. 内存已加载，直接使用
}

const cached = getCachedDictionaries()
if (cached && !expired) {
  return cached // 2. localStorage缓存可用
}

const data = await getAllDictionaries()
return data // 3. 从服务器加载
```

### 2. 降级方案

```javascript
try {
  const data = await getAllDictionaries()
} catch (error) {
  // 加载失败时使用过期缓存
  const expiredCache = getCachedDictionaries(true)
  if (expiredCache) {
    return expiredCache // 降级使用过期数据
  }
}
```

### 3. 异步不阻塞

```javascript
// 字典加载异步进行，不阻塞路由跳转
store.dispatch('dictionary/loadProductionPlanDictionaries').catch(err => {
  console.error('加载字典数据失败:', err)
})
next({ ...to, replace: true }) // 立即跳转
```
